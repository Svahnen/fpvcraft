cmake_minimum_required(VERSION 3.16)
project(fpvcore VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(FPV_BUILD_TESTS "Build unit tests" ON)
option(FPV_BUILD_INPUT "Build the optional fpvinput module (needs SDL2)" ON)
option(FPV_BUILD_VIEWER "Build tools/sim_viewer (needs SDL2 + OpenGL)" ON)
option(FPV_BUILD_SHARED "Build fpvcore as a shared library (for JNI / FFI)" OFF)

# ---- embed data/presets/{curves,drone_*}.json into the library ------------------------------
set(PRESET_DIR ${CMAKE_CURRENT_SOURCE_DIR}/data/presets)
set(EMBED_CPP ${CMAKE_CURRENT_BINARY_DIR}/fpv_embedded.cpp)
set(_ids 3micro 5race 5freestyle 7longrange)
file(READ ${PRESET_DIR}/curves.json _curves)
set(_body "#include <cstddef>\nnamespace fpv {\nextern const char* const kEmbeddedCurves;\nconst char* const kEmbeddedCurves = R\"FPVJSON(${_curves})FPVJSON\";\n")
set(_idlist "")
set(_dlist "")
set(_glist "")
foreach(id ${_ids})
  file(READ ${PRESET_DIR}/drone_${id}.json _d)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PRESET_DIR}/drone_${id}.json)
  string(APPEND _body "static const char* const kP_${id} = R\"FPVJSON(${_d})FPVJSON\";\n")
  string(APPEND _idlist "\"${id}\",")
  string(APPEND _dlist "kP_${id},")
  if(EXISTS ${PRESET_DIR}/drag_${id}.json)   # baked aero tables (tools/bake/bake_drag_generic.py), optional
    file(READ ${PRESET_DIR}/drag_${id}.json _g)
    string(APPEND _body "static const char* const kD_${id} = R\"FPVJSON(${_g})FPVJSON\";\n")
    string(APPEND _glist "kD_${id},")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PRESET_DIR}/drag_${id}.json)
  else()
    string(APPEND _glist "nullptr,")
  endif()
endforeach()
string(APPEND _body "extern const char* const kEmbeddedPresetIds[];\nconst char* const kEmbeddedPresetIds[] = {${_idlist}nullptr};\nextern const char* const kEmbeddedPresets[];\nconst char* const kEmbeddedPresets[] = {${_dlist}nullptr};\nextern const char* const kEmbeddedDrag[];\nconst char* const kEmbeddedDrag[] = {${_glist}nullptr};\nextern const int kEmbeddedPresetCount;\nconst int kEmbeddedPresetCount = 4;\n}\n")
file(GENERATE OUTPUT ${EMBED_CPP} CONTENT "${_body}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PRESET_DIR}/curves.json)

if(FPV_BUILD_SHARED)
  add_library(fpvcore SHARED)
  target_compile_definitions(fpvcore PUBLIC FPV_SHARED)
else()
  add_library(fpvcore STATIC)
endif()
target_sources(fpvcore PRIVATE
  src/fpvcore/api.cpp src/fpvcore/model.cpp src/fpvcore/rates.cpp src/fpvcore/curve.cpp ${EMBED_CPP})
target_include_directories(fpvcore PUBLIC include PRIVATE src/fpvcore third_party)
target_compile_definitions(fpvcore PRIVATE FPV_BUILDING)
set_target_properties(fpvcore PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(fpvcore PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()

if(FPV_BUILD_INPUT)
  find_package(SDL2 QUIET)
  if(NOT SDL2_FOUND)
    find_package(PkgConfig QUIET)
    if(PKG_CONFIG_FOUND)
      pkg_check_modules(SDL2PC QUIET IMPORTED_TARGET sdl2)
    endif()
  endif()
  if(TARGET SDL2::SDL2 OR TARGET PkgConfig::SDL2PC)
    if(TARGET SDL2::SDL2)
      set(FPV_SDL2_TARGET SDL2::SDL2)
    else()
      set(FPV_SDL2_TARGET PkgConfig::SDL2PC)
    endif()
    add_library(fpvinput STATIC src/fpvinput/fpvinput.cpp)
    target_include_directories(fpvinput PUBLIC include PRIVATE third_party)
    target_link_libraries(fpvinput PUBLIC ${FPV_SDL2_TARGET})
  else()
    message(WARNING "SDL2 not found: fpvinput and sim_viewer disabled")
    set(FPV_BUILD_INPUT OFF)
    set(FPV_BUILD_VIEWER OFF)
  endif()
endif()

if(FPV_BUILD_TESTS)
  enable_testing()
  add_executable(fpvcore_tests tests/test_main.cpp tests/test_rates.cpp tests/test_physics.cpp tests/test_convert.cpp)
  target_include_directories(fpvcore_tests PRIVATE third_party src/fpvcore)
  target_link_libraries(fpvcore_tests PRIVATE fpvcore)
  target_compile_definitions(fpvcore_tests PRIVATE FPV_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME fpvcore_tests COMMAND fpvcore_tests)
  if(TARGET fpvinput)
    add_executable(fpvinput_tests tests/test_main.cpp tests/test_input.cpp)
    target_include_directories(fpvinput_tests PRIVATE third_party)
    target_link_libraries(fpvinput_tests PRIVATE fpvinput)
    add_test(NAME fpvinput_tests COMMAND fpvinput_tests)
  endif()
endif()

if(FPV_BUILD_VIEWER AND FPV_BUILD_INPUT)
  find_package(OpenGL QUIET)
  if(OpenGL_FOUND)
    add_executable(sim_viewer tools/sim_viewer/main.cpp)
    target_link_libraries(sim_viewer PRIVATE fpvcore fpvinput OpenGL::GL)
  else()
    message(WARNING "OpenGL not found: sim_viewer disabled")
  endif()
endif()
